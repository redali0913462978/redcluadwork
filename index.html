""" 2D incompressible Navier-Stokes demo: lid-driven cavity. Air in a square box, the top wall slides right (like moving air over a car's surface). Run: python navier_stokes_demo.py """ import numpy as np import matplotlib matplotlib.use("Agg") import matplotlib.pyplot as plt # ---------- 1. Setup: the mesh and fluid properties ---------- nx = ny = 41 # grid points (the "mesh") dx = dy = 1.0 / (nx - 1) # cell size rho, nu = 1.0, 0.1 # density, kinematic viscosity (nu = mu/rho) dt = 0.001 # time step steps, p_iters = 2000, 50 u = np.zeros((ny, nx)) # x-velocity v = np.zeros((ny, nx)) # y-velocity p = np.zeros((ny, nx)) # pressure # ---------- 2. Continuity equation (div u = 0) enforced via pressure ---------- def pressure_source(u, v): """Right-hand side of the pressure Poisson equation.""" b = np.zeros_like(u) b[1:-1, 1:-1] = rho * ( (1 / dt) * ((u[1:-1, 2:] - u[1:-1, :-2]) / (2 * dx) + (v[2:, 1:-1] - v[:-2, 1:-1]) / (2 * dy)) - ((u[1:-1, 2:] - u[1:-1, :-2]) / (2 * dx)) ** 2 - 2 * ((u[2:, 1:-1] - u[:-2, 1:-1]) / (2 * dy) * (v[1:-1, 2:] - v[1:-1, :-2]) / (2 * dx)) - ((v[2:, 1:-1] - v[:-2, 1:-1]) / (2 * dy)) ** 2) return b def solve_pressure(p, b): """Iteratively solve laplacian(p) = b (this is what keeps the flow mass-conserving).""" for _ in range(p_iters): pn = p.copy() p[1:-1, 1:-1] = (((pn[1:-1, 2:] + pn[1:-1, :-2]) * dy**2 + (pn[2:, 1:-1] + pn[:-2, 1:-1]) * dx**2) / (2 * (dx**2 + dy**2)) - dx**2 * dy**2 / (2 * (dx**2 + dy**2)) * b[1:-1, 1:-1]) p[:, -1] = p[:, -2]; p[0, :] = p[1, :] # walls: zero pressure gradient p[:, 0] = p[:, 1]; p[-1, :] = 0 # top: reference pressure return p # ---------- 3. Momentum equation, term by term ---------- for n in range(steps): un, vn = u.copy(), v.copy() p = solve_pressure(p, pressure_source(u, v)) uc, vc = un[1:-1, 1:-1], vn[1:-1, 1:-1] # d(u)/dt = -(u.grad)u - grad(p)/rho + nu*laplacian(u) u[1:-1, 1:-1] = (uc - uc * dt / dx * (uc - un[1:-1, :-2]) # convection (x) - vc * dt / dy * (uc - un[:-2, 1:-1]) # convection (y) - dt / (2 * rho * dx) * (p[1:-1, 2:] - p[1:-1, :-2]) # pressure force + nu * (dt / dx**2 * (un[1:-1, 2:] - 2 * uc + un[1:-1, :-2]) + dt / dy**2 * (un[2:, 1:-1] - 2 * uc + un[:-2, 1:-1]))) # viscosity v[1:-1, 1:-1] = (vc - uc * dt / dx * (vc - vn[1:-1, :-2]) - vc * dt / dy * (vc - vn[:-2, 1:-1]) - dt / (2 * rho * dy) * (p[2:, 1:-1] - p[:-2, 1:-1]) + nu * (dt / dx**2 * (vn[1:-1, 2:] - 2 * vc + vn[1:-1, :-2]) + dt / dy**2 * (vn[2:, 1:-1] - 2 * vc + vn[:-2, 1:-1]))) # ---------- 4. Boundary conditions ---------- u[0, :] = u[:, 0] = u[:, -1] = 0 # no-slip walls u[-1, :] = 1.0 # moving lid (the "wind") v[0, :] = v[-1, :] = v[:, 0] = v[:, -1] = 0 # ---------- 5. Results ---------- speed = np.hypot(u, v) x = y = np.linspace(0, 1, nx) X, Y = np.meshgrid(x, y) fig, ax = plt.subplots(figsize=(6, 5)) c = ax.contourf(X, Y, p, 30, cmap="coolwarm", alpha=.8) ax.streamplot(X, Y, u, v, color=speed, cmap="viridis", density=1.4) fig.colorbar(c, label="pressure") ax.set_title("Lid-driven cavity: streamlines + pressure") fig.savefig("navier_stokes_result.png", dpi=130, bbox_inches="tight") print(f"max speed: {speed.max():.3f}") print(f"pressure range: {p.min():.3f} to {p.max():.3f}") div = np.abs(np.gradient(u, dx, axis=1) + np.gradient(v, dy, axis=0)) print(f"max divergence in the interior (should be small): {div[3:-3, 3:-3].max():.3f}")
